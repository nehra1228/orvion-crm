generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "sqlite"
  url      = env("DATABASE_URL")
}

enum BusinessType { REAL_ESTATE INSURANCE }
enum CompanyStatus { ACTIVE INACTIVE ARCHIVED }
enum BranchStatus { ACTIVE INACTIVE }
enum TeamStatus { ACTIVE INACTIVE }
enum UserStatus { ACTIVE INVITED SUSPENDED INACTIVE }
enum RoleKey { SUPER_ADMIN COMPANY_ADMIN BRANCH_MANAGER DEPARTMENT_MANAGER TEAM_LEADER SENIOR_EXECUTIVE EXECUTIVE HR MARKETING ACCOUNTS QA_MANAGER TELECALLING_MANAGER SUPPORT VIEWER }
enum LeadStatus { NEW CONTACTED INTERESTED FOLLOW_UP BOOKED WON LOST DUPLICATE ON_HOLD }
enum LeadPriority { LOW NORMAL HIGH URGENT }
enum AssignmentMethod { MANUAL ROUND_ROBIN PRIORITY LOAD_BALANCED ESCALATION REASSIGNMENT }
enum RoutingStrategy { ROUND_ROBIN LEAST_LOADED PRIORITY_BASED TEAM_LEADER_MANUAL }
enum ActivityType { CREATED UPDATED ROUTED ASSIGNED REASSIGNED STATUS_CHANGED NOTE_ADDED IMPORTED EXPORTED CALL_LOGGED EMAIL_LOGGED WHATSAPP_LOGGED MEETING_SCHEDULED TASK_CREATED }
enum CallDirection { INBOUND OUTBOUND }
enum CallStatus { QUEUED RINGING CONNECTED COMPLETED MISSED FAILED CANCELED }
enum Disposition { NO_ANSWER CALLBACK REQUESTED_INFO INTERESTED NOT_INTERESTED BOOKED FOLLOW_UP LOST }

model Organization {
  id        String    @id @default(cuid())
  name      String
  code      String    @unique
  status    CompanyStatus @default(ACTIVE)
  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
  companies Company[]
  users     User[]
}

model Company {
  id           String       @id @default(cuid())
  organizationId String
  name         String
  code         String       @unique
  businessType BusinessType
  status       CompanyStatus @default(ACTIVE)
  email        String?
  phone        String?
  website      String?
  address      String?
  settingsJson String?
  createdAt    DateTime     @default(now())
  updatedAt    DateTime     @updatedAt
  organization Organization  @relation(fields: [organizationId], references: [id], onDelete: Cascade)
  branches     Branch[]
  departments  Department[]
  teams        Team[]
  users        UserCompany[]
  leads        Lead[]
  campaigns    Campaign[]
  projects     RealEstateProject[]
  insuranceCategories InsuranceCategory[]
  routingRules LeadRoutingRule[]
  roles        Role[]
  permissions  Permission[]
  calls        CallLog[]
  auditLogs    AuditLog[]
  @@unique([organizationId, name])
  @@index([organizationId, businessType, status])
}

model Branch {
  id        String       @id @default(cuid())
  companyId String
  name      String
  code      String
  city      String?
  state     String?
  status    BranchStatus @default(ACTIVE)
  createdAt DateTime     @default(now())
  updatedAt DateTime     @updatedAt
  company   Company      @relation(fields: [companyId], references: [id], onDelete: Cascade)
  departments Department[]
  teams     Team[]
  users     UserCompany[]
  leads     Lead[]
  routingRules LeadRoutingRule[]
  @@unique([companyId, code])
  @@index([companyId, status])
}

model Department {
  id          String   @id @default(cuid())
  companyId   String
  branchId    String?
  name        String
  code        String
  businessKey String
  active      Boolean  @default(true)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  company     Company  @relation(fields: [companyId], references: [id], onDelete: Cascade)
  branch      Branch?  @relation(fields: [branchId], references: [id], onDelete: SetNull)
  teams       Team[]
  users       UserCompany[]
  leads       Lead[]
  routingRules LeadRoutingRule[]
  @@unique([companyId, code])
  @@index([companyId, branchId, active])
}

model Team {
  id            String     @id @default(cuid())
  companyId     String
  branchId      String?
  departmentId  String
  name          String
  code          String
  status        TeamStatus @default(ACTIVE)
  leaderUserId  String?
  routingOrder  Int        @default(0)
  createdAt     DateTime   @default(now())
  updatedAt     DateTime   @updatedAt
  company       Company    @relation(fields: [companyId], references: [id], onDelete: Cascade)
  branch        Branch?    @relation(fields: [branchId], references: [id], onDelete: SetNull)
  department    Department @relation(fields: [departmentId], references: [id], onDelete: Cascade)
  leader        User?      @relation("TeamLeader", fields: [leaderUserId], references: [id], onDelete: SetNull)
  memberships   TeamMembership[]
  leads         Lead[]
  assignments   LeadAssignment[]
  routingRules  LeadRoutingRule[]
  @@unique([companyId, code])
  @@index([companyId, departmentId, status])
}

model User {
  id           String      @id @default(cuid())
  organizationId String
  username     String      @unique
  email        String      @unique
  fullName     String
  passwordHash String
  status       UserStatus  @default(INVITED)
  phone        String?
  lastLoginAt  DateTime?
  createdAt    DateTime    @default(now())
  updatedAt    DateTime    @updatedAt
  organization Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)
  companies    UserCompany[]
  memberships  TeamMembership[]
  ledTeams     Team[]      @relation("TeamLeader")
  assignedLeads Lead[]     @relation("AssignedExecutive")
  teamLeaderLeads Lead[]   @relation("LeadTeamLeader")
  executiveAssignments LeadAssignment[] @relation("AssignedExecutiveHistory")
  ledAssignments LeadAssignment[] @relation("AssignedBy")
  activities   LeadActivity[]
  auditLogs    AuditLog[]  @relation("AuditActor")
  calls        CallLog[]   @relation("CallExecutive")
  @@index([organizationId, status])
}

model UserCompany {
  id           String      @id @default(cuid())
  userId       String
  companyId    String
  branchId     String?
  departmentId String?
  active       Boolean     @default(true)
  createdAt    DateTime    @default(now())
  user         User        @relation(fields: [userId], references: [id], onDelete: Cascade)
  company      Company     @relation(fields: [companyId], references: [id], onDelete: Cascade)
  branch       Branch?     @relation(fields: [branchId], references: [id], onDelete: SetNull)
  department   Department? @relation(fields: [departmentId], references: [id], onDelete: SetNull)
  roles        UserRole[]
  @@unique([userId, companyId])
  @@index([companyId, branchId, departmentId, active])
}

model Role {
  id          String      @id @default(cuid())
  companyId   String?
  key         RoleKey
  name        String
  description String?
  system      Boolean     @default(false)
  createdAt   DateTime    @default(now())
  company     Company?    @relation(fields: [companyId], references: [id], onDelete: Cascade)
  permissions RolePermission[]
  users       UserRole[]
  @@unique([companyId, key])
  @@index([key])
}

model Permission {
  id          String    @id @default(cuid())
  companyId   String?
  key         String
  module      String
  action      String
  description String?
  createdAt   DateTime  @default(now())
  company     Company?  @relation(fields: [companyId], references: [id], onDelete: Cascade)
  roles       RolePermission[]
  @@unique([companyId, key])
  @@index([module, action])
}

model RolePermission {
  roleId       String
  permissionId String
  granted      Boolean @default(true)
  role         Role    @relation(fields: [roleId], references: [id], onDelete: Cascade)
  permission   Permission @relation(fields: [permissionId], references: [id], onDelete: Cascade)
  @@id([roleId, permissionId])
}

model UserRole {
  userCompanyId String
  roleId        String
  assignedAt    DateTime @default(now())
  userCompany   UserCompany @relation(fields: [userCompanyId], references: [id], onDelete: Cascade)
  role          Role        @relation(fields: [roleId], references: [id], onDelete: Cascade)
  @@id([userCompanyId, roleId])
}

model TeamMembership {

  user            User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  @@unique([teamId, userId])
  @@index([teamId, active, roundRobinOrder])
}

model Lead {
  id                    String        @id @default(cuid())
  companyId             String
  branchId              String?
  departmentId          String?
  teamId                String?
  insuranceCategoryId   String?
  realEstateProjectId   String?
  campaignId            String?
  fullName              String
  email                 String?
  phone                 String
  alternatePhone        String?
  source                String
  category              String?
  subCategory           String?
  city                  String?
  budget                Float?
  priority              LeadPriority  @default(NORMAL)
  status                LeadStatus    @default(NEW)
  subStatus             String?
  assignedExecutiveId   String?
  assignedTeamLeaderId  String?
  routingMethod         AssignmentMethod?
  routedAt              DateTime?
  nextFollowUpAt        DateTime?
  lastActivityAt        DateTime      @default(now())
  duplicateFingerprint  String?
  metadataJson          String?
  createdAt             DateTime      @default(now())
  updatedAt             DateTime      @updatedAt
  company               Company       @relation(fields: [companyId], references: [id], onDelete: Cascade)
  branch                Branch?       @relation(fields: [branchId], references: [id], onDelete: SetNull)
  department            Department?   @relation(fields: [departmentId], references: [id], onDelete: SetNull)
  team                  Team?         @relation(fields: [teamId], references: [id], onDelete: SetNull)
  insuranceCategory     InsuranceCategory? @relation(fields: [insuranceCategoryId], references: [id], onDelete: SetNull)
  realEstateProject     RealEstateProject? @relation(fields: [realEstateProjectId], references: [id], onDelete: SetNull)
  campaign              Campaign?     @relation(fields: [campaignId], references: [id], onDelete: SetNull)
  assignedExecutive     User?         @relation("AssignedExecutive", fields: [assignedExecutiveId], references: [id], onDelete: SetNull)
  assignedTeamLeader    User?         @relation("LeadTeamLeader", fields: [assignedTeamLeaderId], references: [id], onDelete: SetNull)
  activities            LeadActivity[]
  assignments           LeadAssignment[]
  calls                 CallLog[]
  @@index([companyId, departmentId, teamId, status])
  @@index([companyId, assignedExecutiveId, nextFollowUpAt])
  @@index([companyId, source, campaignId])
  @@unique([companyId, duplicateFingerprint])
}

model LeadRoutingRule {
  id                  String          @id @default(cuid())
  companyId           String
  departmentId        String?
  teamId              String?
  branchId            String?
  insuranceCategoryId String?
  projectId           String?
  source              String?
  priority            LeadPriority?
  strategy            RoutingStrategy @default(ROUND_ROBIN)
  rank                Int             @default(100)
  active              Boolean         @default(true)
  createdAt           DateTime        @default(now())
  updatedAt           DateTime        @updatedAt
  company             Company         @relation(fields: [companyId], references: [id], onDelete: Cascade)
  department          Department?     @relation(fields: [departmentId], references: [id], onDelete: SetNull)
  team                Team?           @relation(fields: [teamId], references: [id], onDelete: SetNull)
  branch              Branch?         @relation(fields: [branchId], references: [id], onDelete: SetNull)
  insuranceCategory   InsuranceCategory? @relation(fields: [insuranceCategoryId], references: [id], onDelete: SetNull)
  project             RealEstateProject? @relation(fields: [projectId], references: [id], onDelete: SetNull)
  @@index([companyId, active, rank])
}

model LeadAssignment {
  id            String           @id @default(cuid())
  leadId        String
  teamId        String?
  executiveId   String?
  assignedById  String?
  method        AssignmentMethod
  reason        String?
  active        Boolean          @default(true)
  assignedAt    DateTime         @default(now())
  lead          Lead             @relation(fields: [leadId], references: [id], onDelete: Cascade)
  team          Team?            @relation(fields: [teamId], references: [id], onDelete: SetNull)
  executive     User?            @relation("AssignedExecutiveHistory", fields: [executiveId], references: [id], onDelete: SetNull)
  assignedBy    User?            @relation("AssignedBy", fields: [assignedById], references: [id], onDelete: SetNull)
  @@index([leadId, active])
  @@index([teamId, executiveId, assignedAt])
}

model LeadActivity {

  metadataJson String?
  createdAt   DateTime     @default(now())

}

model CallLog {
  id            String        @id @default(cuid())
  companyId     String
  leadId        String?
  executiveId   String?
  direction     CallDirection
  status        CallStatus    @default(QUEUED)
  disposition   Disposition?
  phoneNumber   String
  startedAt     DateTime?
  endedAt       DateTime?
  durationSec   Int           @default(0)
  recordingUrl  String?
  notes         String?
  tagsJson      String?
  callbackAt    DateTime?
  createdAt     DateTime      @default(now())
  company       Company       @relation(fields: [companyId], references: [id], onDelete: Cascade)
  lead          Lead?         @relation(fields: [leadId], references: [id], onDelete: SetNull)
  executive     User?         @relation("CallExecutive", fields: [executiveId], references: [id], onDelete: SetNull)
  @@index([companyId, status, createdAt])
  @@index([executiveId, callbackAt])
}
