        <motion.div key={String(name)} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: index * .035 }} className={`${glass} rounded-[20px] p-4`}>
          <div className="flex items-center justify-between"><p className="text-[11px] text-slate-400">{String(name)}</p><Icon className="h-3.5 w-3.5 text-cyan-300" /></div>

          <p className="mt-2 text-[10px] text-emerald-300">● Today</p>
        </motion.div>
      ))}
    </section>
    {loading ? <Loading /> : error ? <Failure error={error} retry={load} /> : <div className="grid min-w-0 gap-5 xl:grid-cols-[minmax(0,1fr)_360px]"><section className="space-y-5"><div className="grid gap-5 xl:grid-cols-[minmax(0,1fr)_420px]"><Dialer lead={lead} active={callActive} muted={muted} held={held} clock={clock} autoDial={autoDial} setAutoDial={setAutoDial} start={startCall} end={() => finish("NO_ANSWER")} toggleMute={() => setMuted(!muted)} toggleHold={() => setHeld(!held)} remark={finish} /><ManualActions lead={lead} selectedRemark={selectedRemark} setSelectedRemark={setSelectedRemark} manualNotes={manualNotes} setManualNotes={setManualNotes} saveRemark={saveRemark} savingRemark={savingRemark} /></div><RecentCalls calls={data.recentCalls ?? []} /></section><aside className="space-y-5"><Queue queue={queue} /><FollowUps leads={data.followUps ?? []} /><MiniList title="Missed calls" leads={data.missed ?? []} icon={PhoneOff} /></aside></div>}
  </div><AnimatePresence>{followUp && lead && <FollowUp lead={lead} close={() => setFollowUp(false)} save={async () => { setFollowUp(false); setNotice("Follow-up scheduled."); }} />}</AnimatePresence></main>;
}
